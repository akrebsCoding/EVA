# wyoming-manager – Runtime-Dependencies
# Quelle: PLAN.md §4 (Dependency-Tabelle), verifiziert im Image in P1.T5.
# Ziel-Image: python:3.11-slim (P1.T2) + apt: libsoxr0, libspeex1, netcat-openbsd.
#
# ── WICHTIG: openwakeword braucht --no-deps ────────────────────────────────
#   openwakeword zieht sonst tflite-runtime nach (kein Py3.12-x86_64-Build,
#   PLAN §2.4/§4). Deshalb NICHT blind `pip install -r requirements.txt` in eine
#   Umgebung, in der tflite unerwünscht ist. Die verifizierte Reihenfolge aus
#   P1.T5 lautet:
#
#   pip install onnxruntime "openwakeword==0.6.*" --no-deps speexdsp-ns==0.1.2
#
#   Das Dockerfile (P1.T2) muss openwakeword entsprechend mit --no-deps
#   installieren; speexdsp-ns wird von openwakeword nur für
#   enable_speex_noise_suppression gebraucht (OWW_SPEEX_NS=true).
# ───────────────────────────────────────────────────────────────────────────

fastapi
uvicorn[standard]
websockets

# ── HTTP-Clients: Home Assistant + LLM-Gateway (openai-kompatibel) ───────
httpx

# ── Audio / Math ─────────────────────────────────────────────────────────

soxr                  # Resampling Piper→48 kHz; apt-Paket libsoxr0

# ── mDNS (_emcontroller._tcp.local, P5.T3) ───────────────────────────────
zeroconf

speexdsp-ns==0.1.2    # Speex-NS für enable_speex_noise_suppression; apt libspeex1
